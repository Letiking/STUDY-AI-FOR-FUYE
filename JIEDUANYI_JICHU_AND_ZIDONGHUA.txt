# 列表生成

list1=[i for i in range(10)]

print(list1)

# 列表生成如果偶数保留，否则改为66

list2=[i if i%2==0 else 66 for i in range(10)]

print(list2)

# 列表生成如果偶数保留，否则舍弃，且只保留前3个

list3=[i for i in range(10) if i%2==0][:3]

print(list3)
